#   pip uninstall -y opencv-python opencv-contrib-python opencv-python-headless
#   pip install --no-deps opencv-python-headless
# Verify with: python -c "import cv2; print(cv2.getBuildInformation())"  -> GUI: NONE
# (pip check will then warn that mediapipe wants opencv-contrib-python; that's
# cosmetic — the app makes no cv2 GUI calls and mediapipe imports fine.)
